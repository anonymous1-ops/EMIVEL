gadget SNIISW1
    secrets a,b
    inputs a1:3,b1:3
    outputs c1:3
    randoms rr1:3,re1:2,r1:2
    field 2**7-1

    #noise 1
    #para 3

    a1:=[rr1]
    a2:=[a+a1]
    b1:=[rr2]
    b2:=[b+b1]

    #r1=0

    # refresh
    b1=b1+re1
    b2=b2+re1

    reg=b1
    reg=b2
    
    and0=a1 * b2
    and1=a2 * b1

    xor0=r1 + and0
    xor1=r1 + and1

    reg=and0
    reg=and1
    reg=xor0
    reg=xor1

    out11=a1*b1
    out21=a2*b2

    pc1=out11+xor0
    pc2=out21+xor1
    reg=pc1
    reg=pc2
    c1=pc1
    c2=pc2

end
