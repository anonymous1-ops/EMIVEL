
gadget HPC3+1
	secrets a,b
	inputs a0:2,b0:2
	outputs c0:2
	randoms rr0:2,r0:2,m0:1
	#noise f
	#field 2&&1

	a0:=[rr0]
    a1:=[a^a0]
    b0:=[rr1]
    b1:=[b^b0]

	
	r01_0=r0
	r01_1=r1
	r10_0=r01_0
	r10_1=r01_1

	u01_0=b1^r01_0
	reg=u01_0
	aux1=(~a0)
	aux2=aux1&r01_0
	u01_1=aux2^r01_1
	reg=u01_1
	reg=a0
	aux3=a0&u01_0
	c01=aux3^u01_1

	u10_0=b0^r10_0
	reg=u10_0
	aux4=(~a1)
	aux5=aux4&r10_0
	u10_1=aux5^r10_1
	reg=u10_1
	reg=a1
	aux6=a1&u10_0
	c10=aux6^u10_1

	reg=m0

	t0=a0&b0
	reg=t0
	t1=c01^t0
	c0=t1^m0

	t2=a1&b1
	reg=t2
	t3=c10^t2
	c1=t3^m0


end
