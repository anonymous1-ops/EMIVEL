gadget hpc32
    secrets a,b
    inputs a1:4,b1:4
    outputs c1:4
    randoms ra1:3,rb1:3,r1:7
    field 2**3
    #noise 0.5
    #para 1
    a1:=[ra1]
    a2:=[ra2]
    a3:=[a+a1+a2]

    b1:=[rb1]
    b2:=[rb2]
    b3:=[b+b1+b2]

    #r1=0
    #r2=0
    #r3=0
    #r4=0
    #r5=0
    #r6=0

    a1_not = a1+1
    a2_not = a2+1
    a3_not = a3+1

    # masking
    Up12 = b2 + r1
    reg=Up12
    Upptemp12 = a1_not * r1
    Upp12 = Upptemp12 + r2
    reg=Upp12
    ctemp12 = a1 * Up12
    C12 = ctemp12 + Upp12
    Up21 = b1 + r1
    reg=Up21
    Upptemp21 = a2_not * r1
    Upp21 = Upptemp21 + r2
    reg=Upp21
    ctemp21 = a2 * Up21
    C21 = ctemp21 + Upp21
    Up13 = b3 + r3
    reg=Up13
    Upptemp13 = a1_not * r3
    Upp13 = Upptemp13 + r4
    reg=Upp13
    ctemp13 = a1 * Up13
    C13 = ctemp13 + Upp13
    Up31 = b1 + r3
    reg=Up31
    Upptemp31 = a3_not * r3
    Upp31 = Upptemp31 + r4
    reg=Upp31
    ctemp31 = a3 * Up31
    C31 = ctemp31 + Upp31
    Up23 = b3 + r5
    reg=Up23
    Upptemp23 = a2_not * r5
    Upp23 = Upptemp23 + r6
    reg=Upp23
    ctemp23 = a2 * Up23
    C23 = ctemp23 + Upp23
    Up32 = b2 + r5
    reg=Up32
    Upptemp32 = a3_not * r5
    Upp32 = Upptemp32 + r6
    reg=Upp32
    ctemp32 = a3 * Up32
    C32 = ctemp32 + Upp32
    U11 = a1 * b1
    reg=U11
    U22 = a2 * b2
    reg=U22
    U33 = a3 * b3
    reg=U33

    # outputs
    temp11 = U11 + C12
    c1 = temp11 + C13
    temp21 = U22 + C21
    c2 = temp21 + C23
    temp31 = U33 + C31
    c3 = temp31 + C32

end
