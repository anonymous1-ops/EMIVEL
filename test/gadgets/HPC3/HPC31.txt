gadget hpc31
    secrets a,b
    inputs a1:3,b1:3
    outputs c1:3
    randoms ra1:2,rb1:2,r1:3
    field 2**2
    #noise 1
    #para 4
    a1:=[ra1]
    a2:=[a+a1]

    b1:=[rb1]
    b2:=[b+b1]

    #r1=0
    #r2=0

    a1_not = a1+1
    a2_not = a2+1

    # masking
    Up12 = b2 + r1
    reg=Up12
    Upptemp12 = a1_not * r1
    Upp12 = Upptemp12 + r2
    reg=Upp12
    ctemp12 = a1 * Up12
    C12 = ctemp12 + Upp12
    Up21 = b1 + r1
    reg=Up21
    Upptemp21 = a2_not * r1
    Upp21 = Upptemp21 + r2
    reg=Upp21
    ctemp21 = a2 * Up21
    C21 = ctemp21 + Upp21
    U11 = a1 * b1
    reg=U11
    U22 = a2 * b2
    reg=U22

    # outputs
    c1 = U11 + C12
    c2 = U22 + C21

end
