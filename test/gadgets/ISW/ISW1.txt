gadget ISW1
    secrets a,b
    inputs a1:3,b1:3
    outputs c1:3
    randoms ra1:2,rb1:2,r1:2
    field 2**4
    #noise 1
    #para 4
    a1:=[ra1]
    a2:=[a+a1]

    b1:=[rb1]
    b2:=[b+b1]

    #r1=0

    # Step 2: Compute cross
    and12=a1*b2
    and21=a2*b1
    temp121=r1+and12
    r21=temp121+and21

    # Step 3: Compute output shares c_i
    diag1=a1*b1
    c1=diag1+r1
    diag2=a2*b2
    c2=diag2+r21

end
