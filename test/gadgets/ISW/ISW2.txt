gadget ISW2
    secrets a,b
    inputs a1:4,b1:4
    outputs c1:4
    randoms ra1:3,rb1:3,r1:4
    field 2**2
    #noise 0.5
    #para 1
    a1:=[ra1]
    a2:=[ra2]
    a3:=[a+a1+a2]

    b1:=[rb1]
    b2:=[rb2]
    b3:=[b+b1+b2]

    #r1=0
    #r2=0
    #r3=0

    # Step 2: Compute cross
    and12=a1*b2
    and21=a2*b1
    temp121=r1+and12
    r21=temp121+and21
    and13=a1*b3
    and31=a3*b1
    temp131=r2+and13
    r31=temp131+and31
    and23=a2*b3
    and32=a3*b2
    temp231=r3+and23
    r32=temp231+and32

    # Step 3: Compute output shares c_i
    diag1=a1*b1
    c1_temp1=diag1+r1
    c1=c1_temp1+r2
    diag2=a2*b2
    c2_temp1=diag2+r21
    c2=c2_temp1+r3
    diag3=a3*b3
    c3_temp1=diag3+r31
    c3=c3_temp1+r32

end
