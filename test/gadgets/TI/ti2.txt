gadget ti2
	secrets a,b
	inputs a0:3,b0:3
	outputs c0:3
	randoms rr0:4
	#field 2&&3-1
    para 4
    #noise random,1000


	a0:=[rr0]
    a1:=[rr1]
    a2:=[a^a0^a1]
    b0:=[rr2]
    b1:=[rr3]
    b2:=[b^b0^b1]


	x1y1=a1&b1 
	x1y2=a1&b2 
	x2y1=a2&b1 

	x2y2=a2&b2
	x0y2=a0&b2 
	x2y0=a2&b0

    x0y0=a0&b0
    x0y1=a0&b1
    x1y0=a1&b0

    t0=x1y1^x1y2 
    c0=t0^x2y1 

    t1=x2y2^x0y2
    c1=t1^x2y0

    t2=x0y0^x0y1
    c2=t2^x1y0

end
