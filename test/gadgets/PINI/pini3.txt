gadget PINI3
    secrets a,b
    inputs a1:5,b1:5
    outputs c1:5
    randoms ra1:4,rb1:4,r1:7
    #field 2&&4
    #noise 0.5
    #para 1
    a1:=[ra1]
    a2:=[ra2]
    a3:=[ra3]
    a4:=[a^a1^a2^a3]

    b1:=[rb1]
    b2:=[rb2]
    b3:=[rb3]
    b4:=[b^b1^b2^b3]

    #r1=0
    #r2=0
    #r3=0
    #r4=0
    #r5=0
    #r6=0

    a1_not = ~a1
    a2_not = ~a2
    a3_not = ~a3
    a4_not = ~a4
    # masking
    s12=b2 ^ r1
    p0_12 = a1_not & r1
    p1_12=a1&s12
    Z12=p0_12^p1_12
    s21=b1 ^ r1
    p0_21 = a2_not & r1
    p1_21=a2&s21
    Z21=p0_21^p1_21
    s13=b3 ^ r2
    p0_13 = a1_not & r2
    p1_13=a1&s13
    Z13=p0_13^p1_13
    s31=b1 ^ r2
    p0_31 = a3_not & r2
    p1_31=a3&s31
    Z31=p0_31^p1_31
    s14=b4 ^ r3
    p0_14 = a1_not & r3
    p1_14=a1&s14
    Z14=p0_14^p1_14
    s41=b1 ^ r3
    p0_41 = a4_not & r3
    p1_41=a4&s41
    Z41=p0_41^p1_41
    s23=b3 ^ r4
    p0_23 = a2_not & r4
    p1_23=a2&s23
    Z23=p0_23^p1_23
    s32=b2 ^ r4
    p0_32 = a3_not & r4
    p1_32=a3&s32
    Z32=p0_32^p1_32
    s24=b4 ^ r5
    p0_24 = a2_not & r5
    p1_24=a2&s24
    Z24=p0_24^p1_24
    s42=b2 ^ r5
    p0_42 = a4_not & r5
    p1_42=a4&s42
    Z42=p0_42^p1_42
    s34=b4 ^ r6
    p0_34 = a3_not & r6
    p1_34=a3&s34
    Z34=p0_34^p1_34
    s43=b3 ^ r6
    p0_43 = a4_not & r6
    p1_43=a4&s43
    Z43=p0_43^p1_43
    Z11=a1 & b1
    Z22=a2 & b2
    Z33=a3 & b3
    Z44=a4 & b4

    # outputs
    temp11 = Z11 ^ Z12
    temp12 = temp11 ^ Z13
    temp13 = temp12 ^ Z14
    c1 = temp13
    temp21 = Z22 ^ Z21
    temp22 = temp21 ^ Z23
    temp23 = temp22 ^ Z24
    c2 = temp23
    temp31 = Z33 ^ Z31
    temp32 = temp31 ^ Z32
    temp33 = temp32 ^ Z34
    c3 = temp33
    temp41 = Z44 ^ Z41
    temp42 = temp41 ^ Z42
    temp43 = temp42 ^ Z43
    c4 = temp43

end
