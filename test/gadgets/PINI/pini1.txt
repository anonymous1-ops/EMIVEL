gadget PINI1
    secrets a,b
    inputs a1:3,b1:3
    outputs c1:3
    randoms ra1:2,rb1:2,r1:2
    field 2**4
    #noise 1
    #para 1
    a1:=[ra1]
    a2:=[a+a1]

    b1:=[rb1]
    b2:=[b+b1]

    #r1=0

    a1_not = a1+1
    a2_not = a2+1
    # masking
    s12=b2 + r1
    p0_12 = a1_not * r1
    p1_12=a1*s12
    Z12=p0_12+p1_12
    s21=b1 + r1
    p0_21 = a2_not * r1
    p1_21=a2*s21
    Z21=p0_21+p1_21
    Z11=a1 * b1
    Z22=a2 * b2

    # outputs
    temp11 = Z11 + Z12
    c1 = temp11
    temp21 = Z22 + Z21
    c2 = temp21

end
