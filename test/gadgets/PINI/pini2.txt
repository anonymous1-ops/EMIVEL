gadget PINI2
    secrets a,b
    inputs a1:4,b1:4
    outputs c1:4
    randoms ra1:3,rb1:3,r1:4
    #field 2&&4
    #noise 0.5
    #para 1
    a1:=[ra1]
    a2:=[ra2]
    a3:=[a^a1^a2]

    b1:=[rb1]
    b2:=[rb2]
    b3:=[b^b1^b2]

    #r1=0
    #r2=0
    #r3=0

    a1_not = ~a1
    a2_not = ~a2
    a3_not = ~a3
    # masking
    s12=b2 ^ r1
    p0_12 = a1_not & r1
    p1_12=a1&s12
    Z12=p0_12^p1_12
    s21=b1 ^ r1
    p0_21 = a2_not & r1
    p1_21=a2&s21
    Z21=p0_21^p1_21
    s13=b3 ^ r2
    p0_13 = a1_not & r2
    p1_13=a1&s13
    Z13=p0_13^p1_13
    s31=b1 ^ r2
    p0_31 = a3_not & r2
    p1_31=a3&s31
    Z31=p0_31^p1_31
    s23=b3 ^ r3
    p0_23 = a2_not & r3
    p1_23=a2&s23
    Z23=p0_23^p1_23
    s32=b2 ^ r3
    p0_32 = a3_not & r3
    p1_32=a3&s32
    Z32=p0_32^p1_32
    Z11=a1 & b1
    Z22=a2 & b2
    Z33=a3 & b3

    # outputs
    temp11 = Z11 ^ Z12
    temp12 = temp11 ^ Z13
    c1 = temp12
    temp21 = Z22 ^ Z21
    temp22 = temp21 ^ Z23
    c2 = temp22
    temp31 = Z33 ^ Z31
    temp32 = temp31 ^ Z32
    c3 = temp32

end
