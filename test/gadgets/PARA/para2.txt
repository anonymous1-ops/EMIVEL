gadget para2
	secrets a,b
	inputs a0:3,b0:3
	outputs c0:3
	randoms rr1:5,r0:3
	#field 2&&4
	#noise random
	#para 1

	a0:=[rr1]
	a1:=[rr2]
	a2:=[a^a1^a0]
	b0:=[rr3]
	b1:=[rr4]
	b2:=[b^b1^b0]

	#r0=0
	#r1=0
	#r2=0

	u0=a0&b0
    u1=a1&b1
	u2=a2&b2

    v0=u0^r0
	v1=u1^r1
	v2=u2^r2

    reg=v0
	reg=v1
	reg=v2

	u02=a0&b2
	u10=a1&b0
	u21=a2&b1
	
	v02=v0^u02
	v10=v1^u10
	v21=v2^u21

	reg=v02
	reg=v10
	reg=v21

	u20=a2&b0
	u01=a0&b1
	u12=a1&b2
	
	v20=v02^u20
	v01=v10^u01
	v12=v21^u12

	reg=v20
	reg=v01
	reg=v12

    v11=v20^r1
    v22=v01^r2
	v33=v12^r0

	reg=v11
	reg=v22
	reg=v33

	c0=v11
	c1=v22
	c2=v33
end
