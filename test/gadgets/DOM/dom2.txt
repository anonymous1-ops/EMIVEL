gadget DOM2
    secrets a,b
    inputs a1:4,b1:4
    outputs c1:4
    randoms ra1:3,rb1:3,r1:4
    #field 2&&4
    #noise 0.5
    #para 1
    a1:=[ra1]
    a2:=[ra2]
    a3:=[a^a1^a2]

    b1:=[rb1]
    b2:=[rb2]
    b3:=[b^b1^b2]

    #r1=0
    #r2=0
    #r3=0

    # AND
    u11=a1&b1
    u12=a1&b2
    u13=a1&b3
    u21=a2&b1
    u22=a2&b2
    u23=a2&b3
    u31=a3&b1
    u32=a3&b2
    u33=a3&b3
    reg=u12
    reg=u13
    reg=u21
    reg=u23
    reg=u31
    reg=u32

    # masking
    v12=u12^r1
    v21=u21^r1
    v13=u13^r2
    v31=u31^r2
    v23=u23^r3
    v32=u32^r3
    reg=v12
    reg=v13
    reg=v21
    reg=v23
    reg=v31
    reg=v32

    # outputs
    sum10=v12^v13
    sum20=v21^v23
    sum30=v31^v32
    reg=sum10
    reg=sum20
    reg=sum30

    c1=sum10^u11
    c2=sum20^u22
    c3=sum30^u33

end
