gadget IPM2
    secrets a,b
    inputs a1:4,b1:4
    outputs c1:4
    randoms ra2:4,rb2:4,r1:4
    field 2**2
    #noise 0.5
    #para 1
    
    l2 = 2
    l3 = 3

    a2:=[ra2]
    a3:=[ra3]
    a1:=[a+(l2*a2+l3*a3)]

    b2:=[rb2]
    b3:=[rb3]
    b1:=[b+(l2*b2+l3*b3)]

    #r1=0
    #r2=0
    #r3=0

    # matrix T
    T11=a1*b1
    t12=a1*b2
    T12=t12*l2
    t13=a1*b3
    T13=t13*l3
    T21=a2*b1
    t22=a2*b2
    T22=t22*l2
    t23=a2*b3
    T23=t23*l3
    T31=a3*b1
    t32=a3*b2
    T32=t32*l2
    t33=a3*b3
    T33=t33*l3

    # matrix U
    U21=(-r1)*(1/l2)
    U23=r3*(1/l2)
    U31=(-r2)*(1/l3)
    U32=(-r3)*(1/l3)

    # matrix V
    V12=T12+r1
    V13=T13+r2
    V21=T21+U21
    V23=T23+U23
    V31=T31+U31
    V32=T32+U32

    # outputs
    sum_c11=T11+V12
    c1=sum_c11+V13
    sum_c21=V21+T22
    c2=sum_c21+V23
    sum_c31=V31+V32
    c3=sum_c31+T33

end
