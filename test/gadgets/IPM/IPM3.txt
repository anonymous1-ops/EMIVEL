gadget IPM3
    secrets a,b
    inputs a1:5,b1:5
    outputs c1:5
    randoms ra2:5,rb2:5,r1:7
    field 2**2
    #noise 0.5
    #para 1

    l2 = 1
    l3 = 2
    l4 = 3

    a2:=[ra2]
    a3:=[ra3]
    a4:=[ra4]
    a1:=[a+(l2*a2+l3*a3+l4*a4)]

    b2:=[rb2]
    b3:=[rb3]
    b4:=[rb4]
    b1:=[b+(l2*b2+l3*b3+l4*b4)]

    #r1=0
    #r2=0
    #r3=0
    #r4=0
    #r5=0
    #r6=0

    # matrix T
    T11=a1*b1
    t12=a1*b2
    T12=t12*l2
    t13=a1*b3
    T13=t13*l3
    t14=a1*b4
    T14=t14*l4
    T21=a2*b1
    t22=a2*b2
    T22=t22*l2
    t23=a2*b3
    T23=t23*l3
    t24=a2*b4
    T24=t24*l4
    T31=a3*b1
    t32=a3*b2
    T32=t32*l2
    t33=a3*b3
    T33=t33*l3
    t34=a3*b4
    T34=t34*l4
    T41=a4*b1
    t42=a4*b2
    T42=t42*l2
    t43=a4*b3
    T43=t43*l3
    t44=a4*b4
    T44=t44*l4

    # matrix U
    U21=(-r1)*(1/l2)
    U23=r4*(1/l2)
    U24=r5*(1/l2)
    U31=(-r2)*(1/l3)
    U32=(-r4)*(1/l3)
    U34=r6*(1/l3)
    U41=(-r3)*(1/l4)
    U42=(-r5)*(1/l4)
    U43=(-r6)*(1/l4)

    # matrix V
    V12=T12+r1
    V13=T13+r2
    V14=T14+r3
    V21=T21+U21
    V23=T23+U23
    V24=T24+U24
    V31=T31+U31
    V32=T32+U32
    V34=T34+U34
    V41=T41+U41
    V42=T42+U42
    V43=T43+U43

    # outputs
    sum_c11=T11+V12
    sum_c12=sum_c11+V13
    c1=sum_c12+V14
    sum_c21=V21+T22
    sum_c22=sum_c21+V23
    c2=sum_c22+V24
    sum_c31=V31+V32
    sum_c32=sum_c31+T33
    c3=sum_c32+V34
    sum_c41=V41+V42
    sum_c42=sum_c41+V43
    c4=sum_c42+T44

end
