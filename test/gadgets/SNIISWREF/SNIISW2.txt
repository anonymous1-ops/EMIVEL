gadget SNIISWREF2
    secrets a,b
    inputs a1:4,b1:4
    outputs c1:4
    randoms ra1:3,rb1:3,re1:3,r1:4
    field 2**4
    #noise 0.5
    #para 1
    a1:=[ra1]
    a2:=[ra2]
    a3:=[a+a1+a2]

    b1:=[rb1]
    b2:=[rb2]
    b3:=[b+b1+b2]

    #r1=0
    #r2=0
    #r3=0

    # refresh
    # optimized glitch-robust SNI refresh
    t1 = re1 + re2
    reg = t1
    b1 = b1 + re1
    reg = b1
    b2 = b2 + re2
    reg = b2
    b3 = b3 + t1
    reg = b3

    # AND
    and11=a1 * b1
    and12=a1 * b2
    and13=a1 * b3
    and21=a2 * b1
    and22=a2 * b2
    and23=a2 * b3
    and31=a3 * b1
    and32=a3 * b2
    and33=a3 * b3

    # masking
    u12=r1 + and12
    u21=r1 + and21
    u13=r2 + and13
    u31=r2 + and31
    u23=r3 + and23
    u32=r3 + and32

    reg = and11
    reg = and12
    reg = and13
    reg = and21
    reg = and22
    reg = and23
    reg = and31
    reg = and32
    reg = and33
    reg = u12
    reg = u21
    reg = u13
    reg = u31
    reg = u23
    reg = u32

    # outputs
    temp11 = and11 + u12
    final1 = temp11 + u13
    temp21 = and22 + u21
    final2 = temp21 + u23
    temp31 = and33 + u31
    final3 = temp31 + u32
    reg = final1
    reg = final2
    reg = final3
    c1=final1
    c2=final2
    c3=final3
end
