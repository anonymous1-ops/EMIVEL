gadget SNIISWREF3
    secrets a,b
    inputs a1:5,b1:5
    outputs c1:5
    randoms ra1:4,rb1:4,re1:5,r1:7
    field 2**4
    #noise 0.5
    #para 1
    a1:=[ra1]
    a2:=[ra2]
    a3:=[ra3]
    a4:=[a+a1+a2+a3]

    b1:=[rb1]
    b2:=[rb2]
    b3:=[rb3]
    b4:=[b+b1+b2+b3]

    #r1=0
    #r2=0
    #r3=0
    #r4=0
    #r5=0
    #r6=0

    # refresh
    # optimized glitch-robust SNI refresh
    t1 = re1 + re2
    reg = t1
    t2 = re2 + re3
    reg = t2
    t3 = re3 + re4
    reg = t3
    t4 = re4 + re1
    reg = t4
    b1 = b1 + t1
    reg = b1
    b2 = b2 + t2
    reg = b2
    b3 = b3 + t3
    reg = b3
    b4 = b4 + t4
    reg = b4

    # AND
    and11=a1 * b1
    and12=a1 * b2
    and13=a1 * b3
    and14=a1 * b4
    and21=a2 * b1
    and22=a2 * b2
    and23=a2 * b3
    and24=a2 * b4
    and31=a3 * b1
    and32=a3 * b2
    and33=a3 * b3
    and34=a3 * b4
    and41=a4 * b1
    and42=a4 * b2
    and43=a4 * b3
    and44=a4 * b4

    # masking
    u12=r1 + and12
    u21=r1 + and21
    u13=r2 + and13
    u31=r2 + and31
    u14=r3 + and14
    u41=r3 + and41
    u23=r4 + and23
    u32=r4 + and32
    u24=r5 + and24
    u42=r5 + and42
    u34=r6 + and34
    u43=r6 + and43

    reg = and11
    reg = and12
    reg = and13
    reg = and14
    reg = and21
    reg = and22
    reg = and23
    reg = and24
    reg = and31
    reg = and32
    reg = and33
    reg = and34
    reg = and41
    reg = and42
    reg = and43
    reg = and44
    reg = u12
    reg = u21
    reg = u13
    reg = u31
    reg = u14
    reg = u41
    reg = u23
    reg = u32
    reg = u24
    reg = u42
    reg = u34
    reg = u43

    # outputs
    temp11 = and11 + u12
    temp12 = temp11 + u13
    final1 = temp12 + u14
    temp21 = and22 + u21
    temp22 = temp21 + u23
    final2 = temp22 + u24
    temp31 = and33 + u31
    temp32 = temp31 + u32
    final3 = temp32 + u34
    temp41 = and44 + u41
    temp42 = temp41 + u42
    final4 = temp42 + u43
    reg = final1
    reg = final2
    reg = final3
    reg = final4
    c1=final1
    c2=final2
    c3=final3
    c4=final4
end
