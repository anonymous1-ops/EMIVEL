gadget hpc22
    secrets a,b
    inputs a1:4,b1:4
    outputs c1:4
    randoms ra1:3,rb1:3,r1:4
    field 2**4
    #noise 0.5
    #para 1
    a1:=[ra1]
    a2:=[ra2]
    a3:=[a+a1+a2]

    b1:=[rb1]
    b2:=[rb2]
    b3:=[b+b1+b2]

    #r1=0
    #r2=0
    #r3=0

    # masking
    a1_not = a1+1
    a2_not = a2+1
    a3_not = a3+1
    u12 = a1_not*r1
    u21 = a2_not*r1
    v12 = b2+r1
    v21 = b1+r1
    u13 = a1_not*r2
    u31 = a3_not*r2
    v13 = b3+r2
    v31 = b1+r2
    u23 = a2_not*r3
    u32 = a3_not*r3
    v23 = b3+r3
    v32 = b2+r3
    a1b1 = a1*b1
    a2b2 = a2*b2
    a3b3 = a3*b3
    reg=u11
    reg=u12
    reg=v12
    reg=u13
    reg=v13
    reg=u21
    reg=v21
    reg=u22
    reg=u23
    reg=v23
    reg=u31
    reg=v31
    reg=u32
    reg=v32
    reg=u33

    # outputs
    a1v12 = a1*v12
    reg=a1v12
    a1v13 = a1*v13
    reg=a1v13
    u12_plus = u12+a1v12
    u13_plus = u13+a1v13
    o11 = a1b1+u12_plus
    c1 = o11+u13_plus
    a2v21 = a2*v21
    reg=a2v21
    a2v23 = a2*v23
    reg=a2v23
    u21_plus = u21+a2v21
    u23_plus = u23+a2v23
    o21 = a2b2+u21_plus
    c2 = o21+u23_plus
    a3v31 = a3*v31
    reg=a3v31
    a3v32 = a3*v32
    reg=a3v32
    u31_plus = u31+a3v31
    u32_plus = u32+a3v32
    o31 = a3b3+u31_plus
    c3 = o31+u32_plus

end
