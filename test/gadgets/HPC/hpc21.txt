gadget hpc21
    secrets a,b
    inputs a1:3,b1:3
    outputs c1:3
    randoms ra1:2,rb1:2,r1:2
    field 2**4
    #noise 1
    #para 4
    a1:=[ra1]
    a2:=[a+a1]

    b1:=[rb1]
    b2:=[b+b1]

    #r1=0

    # masking
    a1_not = a1+1
    a2_not = a2+1
    u12 = a1_not*r1
    u21 = a2_not*r1
    v12 = b2+r1
    v21 = b1+r1
    a1b1 = a1*b1
    a2b2 = a2*b2
    #reg=u11
    #reg=u12
    #reg=v12
    #reg=u21
    #reg=v21
    #reg=u22

    # outputs
    a1v12 = a1*v12
    #reg=a1v12
    u12_plus = u12-a1v12
    c1 = a1b1+u12_plus
    a2v21 = a2*v21
    #reg=a2v21
    u21_plus = u21-a2v21
    c2 = a2b2+u21_plus

end
